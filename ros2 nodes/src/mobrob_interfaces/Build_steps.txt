### Build Steps for "mobrob_interfaces" 
# Updated 2024-10-07

cd ~/ros2_ws/src

# Create the package "mobrob_interfaces"
ros2 pkg create mobrob_interfaces --build-type ament_cmake --license Apache-2.0 --dependencies std_msgs geometry_msgs sensor_msgs nav_msgs map_msgs builtin_interfaces

cd mobrob_interfaces
mkdir msg

## Paste in all the Message files *.msg

## CMakeLists: Add stuff: 

# Should already be there due to inclusion in the "pkg create":  command
# find dependencies  
find_package(ament_cmake REQUIRED)
find_package(example_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(map_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)

# Added stuff: 
find_package(rosidl_default_generators REQUIRED)
set(msg_files
  "msg/ME439MotorCommands.msg"
  "msg/ME439PathSpecs.msg"
  "msg/ME439SensorsRaw.msg"
  "msg/ME439SensorsProcessed.msg"
  "msg/ME439WaypointXY.msg"
  "msg/ME439WheelAngles.msg"
  "msg/ME439WheelDisplacements.msg"
  "msg/ME439WheelSpeeds.msg"
)
rosidl_generate_interfaces(${PROJECT_NAME}
  ${msg_files}
  DEPENDENCIES builtin_interfaces
)
ament_export_dependencies(rosidl_default_runtime)


---
cd ~/ros2_ws
colcon build --symlink-install --packages-up-to mobrob_interfaces 
# Takes about 90 seconds
